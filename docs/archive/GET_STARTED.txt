🔌 QUICK API TESTS (with curl):
════════════════════════════════════════════════════════════════════════════════

CREATE WAREHOUSE:
  curl -X POST http://localhost:3000/api/warehouses \
    -H "Content-Type: application/json" \
    -d '{"name":"Main","location":"Downtown","isDefault":true}'

CREATE INVENTORY:
  curl -X POST http://localhost:3000/api/inventory \
    -H "Content-Type: application/json" \
    -d '{"productId":"P1","warehouseId":"W1","quantity":100,"reorderLevel":20}'

ADJUST STOCK:
  curl -X POST http://localhost:3000/api/inventory/adjust \
    -H "Content-Type: application/json" \
    -d '{"inventoryId":"I1","adjustmentType":"add","quantity":50,"reason":"PO#123"}'

TRANSFER STOCK:
  curl -X POST http://localhost:3000/api/inventory/transfer \
    -H "Content-Type: application/json" \
    -d '{"fromInventoryId":"I1","toInventoryId":"I2","quantity":25,"reason":"rebalance"}'

LIST INVENTORY:
  curl http://localhost:3000/api/inventory
