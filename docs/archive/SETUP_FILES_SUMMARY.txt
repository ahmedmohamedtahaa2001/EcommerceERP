📝 COLLECTION SCHEMAS AT A GLANCE:
═════════════════════════════════════════════════════════════════════════════

COMPANIES:
  name (string), email (string), industry (string), website (string),
  country (string), active (boolean), subscription (enum)

WAREHOUSES:
  companyId, name, location, address, city, state, zipCode, country,
  isDefault (boolean), capacity (number), currentUtilization (number)

PRODUCTS:
  companyId, sku, name, description, category, price, cost, margin,
  weight, dimensions (map), active (boolean), barcode

INVENTORY:
  companyId, productId, warehouseId, quantity, reserved, available,
  reorderLevel, reorderQuantity, lastRestocked, lastChecked

STOCK_MOVEMENTS (Immutable):
  companyId, inventoryId, productId, warehouseId, type (enum),
  quantityBefore, quantityAfter, quantityChanged, reason, notes

USERS:
  companyId, email, name, role (enum), department, phone, avatar,
  active (boolean), lastLogin, permissions (array)

AUDIT_LOGS (Immutable):
  companyId, userId, action, resource, resourceId, changes (map), ipAddress
